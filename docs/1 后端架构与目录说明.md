#### 目录解析
backend/
├── docs/                        # 文档目录
├── package.json                 # 记录项目名称、启动脚本 (npm run dev) 及依赖包列表
├── package-lock.json            # 依赖版本锁：第三方包的具体版本
├── tsconfig.json                # TypeScript 编译配置文件：规定 TS 代码如何转译为 JS
└── src/                         # 源代码核心目录
    │
    ├── app.ts                   # 全局总入口,挂载所有路由、初始化全局中间件、启动 Express 监听端口
    │
    ├── db/                      # 数据库层
    │   └── index.ts             # 金仓数据库连接池配置：负责与底层数据库建立稳定、高效的连接
    │
    ├── middlewares/             # 中间件层 (系统拦截器)
    │   └── auth.ts              # 核心安全鉴权：拦截请求，校验 Header 中的 JWT Token，解析出登录用户的 id 和角色
    │
    ├── routes/                  # 路由层：定义 API 地址，并将请求转交给对应的 Controller处理
    │   ├── application.ts       # 证明申请与审批模块路由 (/api/applications)
    │   ├── auth.ts              # 登录与权限模块路由 (/api/auth)
    │   ├── knowledge.ts         # 政策知识库问答路由 (/api/knowledge)
    │   ├── notice.ts            # 全校通知分发与统计路由 (/api/notices)
    │   └── process.ts           # 党团流程节点流转路由 (/api/process)
    │
    ├── controllers/                 # 控制层：负责校验前端参数，并统一组装返回 { code, message, data }
    │   ├── applicationController.ts # 申请与审批
    │   ├── authController.ts        # 接收账号密码，校验成功后签发下发 Token
    │   ├── knowledgeController.ts   # 接收学生的问句参数，判空后转交智能匹配服务
    │   ├── noticeController.ts      # 发布通知，标记已读以及管理员查看已读统计等
    │   └── processController.ts     # 党团流程阶段以及进度
    │
    └── services/                    # 业务层：负责极其复杂的业务计算与人大金仓 SQL 执行
        ├── applicationService.ts    
        ├── jobService.ts            
        ├── knowledgeService.ts      
        ├── noticeService.ts         
        └── processService.ts
